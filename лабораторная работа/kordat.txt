RESTART=0;
TITLE='VD.035.004 CYLINDER STEP + SIN (CONVERGENCE)';

! ==========================================================
! РЕЖИМ: 0 = скачок, 1 = гармоника
! ==========================================================
MODE  = 0;
OMEGA = 1.0;
PI = 3.14159265358979;
PERIOD = 2.*PI/OMEGA;

! ==========================================================
! ПАРАМЕТРЫ СХОДИМОСТИ (менять здесь!)
! ==========================================================
K_RAD  = 21;        ! число узлов по радиусу (5, 11, 21, 41)
DT_MAX_USER = 0.001;  ! шаг по времени (1e-2, 5e-3, 1e-3, 5e-4)

! ==========================================================
! 1. Исходные данные
! ==========================================================
R_cyl = 0.0025;
LAM_ST = 30.;
RHO_ST = 7800.;
C_ST = 500.;
CR_ST = RHO_ST * C_ST;

T_0 = 293.15;
T_F = 473.15;
DT_T = T_F - T_0;
A_IN = 0.5*DT_T;

ALPHA = 1000.;
A_DIF = LAM_ST / CR_ST;

! ==========================================================
! 2. Ключи основной процедуры — зависят от MODE
! ==========================================================
IF MODE == 0 THEN
   FIN_TIM = 50.;
   DT_MAX  = DT_MAX_USER;
   DT_OUT  = 0.1;
ELSE
   FIN_TIM = 20.*PERIOD;
   DT_MAX  = MIN(DT_MAX_USER, PERIOD/200.);
   DT_OUT  = PERIOD/100.;
ENDIF
DT_SAV  = 0.;

! ==========================================================
! 3. Глобальные переменные
! ==========================================================
FO_VAL  = 0.;

! --- THETA: массив размером K_RAD ---
IF MODE == 0 THEN
   THETA(1:K_RAD) = 1.0;
ELSE
   THETA(1:K_RAD) = 0.0;
ENDIF

I_LOOP  = 0;
Q_SURF  = 0.;
Q_LIN   = 0.;
T_ENV   = T_0;
T_W     = T_0;
T_C     = T_0;
T_ENV_IN = T_0;
E_ACC   = 0.;

! ==========================================================
! 4. LAYOUT
! ==========================================================
LAYOUT
HCS1(1)/2 - BHEAT1;
HCS1(1)/1 - BHEAT2;
END

! ==========================================================
! 5. DATA
! ==========================================================
CALL HCS1;
CALL BHEAT1;
CALL BHEAT2;

! ==========================================================
! 6. SET
! ==========================================================
SET STEP_ENV;
SET CALC_THETA;
SET CALC_ENERGY;

! ==========================================================
! 7. OUT
! ==========================================================
OUT PROFILES;
END

! ==========================================================
! DATA HCS1
! ==========================================================
DATA HCS1
N = 1;
KL = 1;
K = K_RAD;                     ! < число узлов из K_RAD
TYPE = 0;
COOR = 1;
XL(1:2) = 0., R_cyl;
X(1:K_RAD) = LINA(0., R_cyl, K_RAD);   ! < K_RAD узлов
DFZ(1) = 1.0;
B = 1.0;
LAM = LAM_ST;
CR  = CR_ST;
END

! ==========================================================
! DATA BHEAT1 / BHEAT2
! ==========================================================
DATA BHEAT1
TYPE = 3;
BCOND(1:2) = ALPHA, T_0;
END

DATA BHEAT2
TYPE = 2;
BCOND(1:2) = 0., 0.;
END

! ==========================================================
! EVENT: ГУ — скачок ИЛИ гармоника
! ==========================================================
EVENT STEP_ENV
IF MODE == 0 THEN
   BCOND.BHEAT1(2) = T_F;
   T_ENV_IN = T_F;
ELSE
   T_ENV_IN = T_0 + A_IN*SIN(OMEGA*TAU);
   BCOND.BHEAT1(2) = T_ENV_IN;
ENDIF
END

! ==========================================================
! EVENT: ?, потоки, T_C
! ==========================================================
EVENT CALC_THETA
FO_VAL = A_DIF*TAU/(R_cyl**2);
T_ENV = BCOND.BHEAT1(2);
T_W   = TW.HCS1(2,1);
T_C   = T.HCS1(1,1);
Q_SURF = ALPHA * (T_ENV - T_W);
Q_LIN  = Q_SURF * 2. * PI * R_cyl;
IF MODE == 0 THEN
   DO I_LOOP=1,K_RAD;           ! < K_RAD
     THETA(I_LOOP) = (T.HCS1(I_LOOP,1) - T_F)/(T_0 - T_F);
   ENDDO;
ELSE
   DO I_LOOP=1,K_RAD;           ! < K_RAD
     THETA(I_LOOP) = (T.HCS1(I_LOOP,1) - T_0)/A_IN;
   ENDDO;
ENDIF
END

! ==========================================================
! EVENT: энергия (только для скачка)
! ==========================================================
EVENT CALC_ENERGY
IF MODE == 0 [ E_ACC = E_ACC + Q_LIN * DT; ]
END

! ==========================================================
! OUTPUT PROFILES
! ==========================================================
OUTPUT PROFILES
WRITE T_ENV_IN, T_C, T_W, Q_SURF, Q_LIN, E_ACC;
WRITE THETA(1:K_RAD), T.HCS1(1:K_RAD,1), FO_VAL;
END
